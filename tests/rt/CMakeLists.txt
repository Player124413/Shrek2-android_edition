# Тесты рантайма рекомпилятора без игры
add_executable(rep_test rep_test.c)
target_include_directories(rep_test PRIVATE ${PROJECT_SOURCE_DIR})
target_link_libraries(rep_test PRIVATE snx_runtime)
add_executable(fist_test fist_test.c)
target_include_directories(fist_test PRIVATE ${PROJECT_SOURCE_DIR})
target_link_libraries(fist_test PRIVATE snx_runtime)
add_executable(wfile_test wfile_test.c $<TARGET_OBJECTS:snx_hle>)
target_include_directories(wfile_test PRIVATE ${PROJECT_SOURCE_DIR})
target_link_libraries(wfile_test PRIVATE snx_hle snx_runtime)
add_executable(crand_test crand_test.c)
target_include_directories(crand_test PRIVATE ${PROJECT_SOURCE_DIR})
target_link_libraries(crand_test PRIVATE snx_runtime)
add_executable(dinput_test dinput_test.c $<TARGET_OBJECTS:snx_hle>)
target_include_directories(dinput_test PRIVATE ${PROJECT_SOURCE_DIR})
target_link_libraries(dinput_test PRIVATE snx_hle snx_runtime)
add_executable(ini_test ini_test.c ${PROJECT_SOURCE_DIR}/host/boot/config.c)
target_include_directories(ini_test PRIVATE ${PROJECT_SOURCE_DIR})

foreach(test rep_test fist_test crand_test dinput_test)
  add_test(NAME ${test} COMMAND ${test})
endforeach()
add_test(NAME wfile_test COMMAND wfile_test "${CMAKE_CURRENT_BINARY_DIR}/wfile-data")
add_test(NAME wfile_no_cache_test COMMAND wfile_test "${CMAKE_CURRENT_BINARY_DIR}/wfile-no-cache-data")
set_tests_properties(wfile_no_cache_test PROPERTIES ENVIRONMENT "SNX_FILE_CACHE_MB=0")
add_test(NAME ini_test COMMAND ini_test "${CMAKE_CURRENT_BINARY_DIR}/ini-data")
set_tests_properties(rep_test fist_test crand_test dinput_test wfile_test wfile_no_cache_test ini_test PROPERTIES TIMEOUT 60)
